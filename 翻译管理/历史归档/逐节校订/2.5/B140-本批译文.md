## 5.2a. 1/m 界面外侧 ($x = -2n h_m\tan\theta_{m,SF}, z = 0^-$) 的反射 SFG

（同时使用 $1/\cos\theta = \cos\theta + \sin\theta\tan\theta$ 的关系式，反射率和透射率的下标 SF 省略后，将 p 偏振或 s 偏振的电场振幅记为 $E_t$，极化的 x、y、z 分量记为 $P_a$。）

也使用了 $k_{m,SF}\sin\theta_{m,SF} - k_{m,vis}\sin\theta_{m,vis} - k_{m,ir}\sin\theta_{m,ir} = 0$ 的关系式进行了整理。

$$
\begin{aligned}
E_{1p}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,x}\chi_{x\beta\gamma}+(1-r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,z}\chi_{z\beta\gamma}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5.13a}
$$

$$
\begin{aligned}
E_{1s}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,s,y}\chi_{y\beta\gamma}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5.13b}
$$

当表面形成完整薄膜时，对于构成该薄膜（折射率为 $m'$）的分子的 SFG，需要考虑薄膜内部的多次反射。

$$
\begin{aligned}
E^-_{1p}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{(1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p})[(1+r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,x}\chi_{x\beta\gamma}+(1-r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,z}\chi_{z\beta\gamma}]}{1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p}+r_{m2,\mathrm{SF},p}(r_{1m',\mathrm{SF},p}+r_{m'm,\mathrm{SF},p})e^{2i\beta_{m,\mathrm{SF}}h_m}}
\end{aligned}\tag{5.14a}
$$

$$
\begin{aligned}
E^-_{1s}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{(1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s})[(1+r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,s,y}\chi_{y\beta\gamma}]}{1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s}+r_{m2,\mathrm{SF},s}(r_{1m',\mathrm{SF},s}+r_{m'm,\mathrm{SF},s})e^{2i\beta_{m,\mathrm{SF}}h_m}}
\end{aligned}\tag{5.14b}
$$
