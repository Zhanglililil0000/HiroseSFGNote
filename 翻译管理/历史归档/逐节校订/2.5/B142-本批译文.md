## 5.3 透射 SFG 的电场

在考虑透射 SFG 时，除了上述反射 SFG 中已考虑的部分外，还必须纳入 $x = 0 \sim h_m\tan\theta_{m,SF}$ 范围（以及 m/2 界面外侧）的贡献。

### 5.3a. 1/m 界面外侧 ($x = -2n h_m\tan\theta_{m,SF}, z = 0^-$) 的透射 SFG

$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{SF},p}t_{m2,\mathrm{SF},p}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}(L^{+}_{m/1,p,x}\chi_{x\beta\gamma}+L^{+}_{m/1,p,z}\chi_{z\beta\gamma})
\end{aligned}\tag{5.20a}
$$

$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{SF},s}t_{m2,\mathrm{SF},s}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}L^{+}_{m/1,s,y}\chi_{y\beta\gamma}
\end{aligned}\tag{5.20b}
$$

当发光层为薄膜 $m'$ 时：

$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{t_{m2,\mathrm{SF},p}(1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p})e^{i\beta_{m,\mathrm{SF}}h_m}}{1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p}+r_{m2,\mathrm{SF},p}(r_{mm',\mathrm{SF},p}+r_{m'1,\mathrm{SF},p})e^{2i\beta_{m,\mathrm{SF}}h_m}}\mathclose{)}(L^{+}_{1/m,p,x}\chi_{x\beta\gamma}+L^{+}_{1/m,p,z}\chi_{z\beta\gamma})
\end{aligned}\tag{5.21a}
$$

$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{t_{m2,\mathrm{SF},s}(1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s})e^{i\beta_{m,\mathrm{SF}}h_m}}{1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s}+r_{m2,\mathrm{SF},s}(r_{mm',\mathrm{SF},s}+r_{m'1,\mathrm{SF},s})e^{2i\beta_{m,\mathrm{SF}}h_m}}\mathclose{)}L^{+}_{1/m,s,y}\chi_{y\beta\gamma}
\end{aligned}\tag{5.21b}
$$

### 5.3b. 1/m 界面内侧 ($x = -2n h_m\tan\theta_{m,SF}, z = 0^+$) 的透射 SFG

$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{m2,\mathrm{SF},p}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}(L^{+}_{1/m,p,x}\chi_{x\beta\gamma}+L^{+}_{1/m,p,z}\chi_{z\beta\gamma})
\end{aligned}\tag{5.22a}
$$

$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{m2,\mathrm{SF},s}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}L^{+}_{1/m,s,y}\chi_{y\beta\gamma}
\end{aligned}\tag{5.22b}
$$

### 5.3c. m/2 界面外侧 ($x = -(2n + 1)h_m\tan\theta_{m,SF}, z = h_m^+$) 的透射 SFG

$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}t_{1m,\mathrm{ir},\gamma}t_{m2,\mathrm{vis},\beta}t_{m2,\mathrm{ir},\gamma}e^{i(\beta_{m,\mathrm{ir}}+\beta_{m,\mathrm{vis}})h_m}}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,wvis,\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1-r_{1m,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,p,x}\chi_{x\beta\gamma}+(1+r_{1m,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,p,z}\chi_{z\beta\gamma}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5.23a}
$$

$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}t_{1m,\mathrm{ir},\gamma}t_{m2,\mathrm{vis},\beta}t_{m2,\mathrm{ir},\gamma}e^{i(\beta_{m,\mathrm{ir}}+\beta_{m,\mathrm{vis}})h_m}}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,wvis,\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1-r_{1m,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,s,y}\chi_{y\beta\gamma}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5.23b}
$$

### 5.3d. m/2 界面内侧 ($x = -(2n + 1)h_m\tan\theta_{m,SF}, z = h_m^-$) 的透射 SFG

$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma}e \frac{t_{1m,\mathrm{vis},\beta}t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{vis},\beta})(1+r_{m2,\mathrm{ir},\gamma})e^{i(\beta_{m,\mathrm{ir}}+\beta_{m,\mathrm{vis}})h_m}}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m1,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,p,x}\chi_{x\beta\gamma}+(-1+r_{m1,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,p,z}\chi_{z\beta\gamma}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5.24a}
$$

$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma}e \frac{t_{1m,\mathrm{vis},\beta}t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{vis},\beta})(1+r_{m2,\mathrm{ir},\gamma})e^{i(\beta_{m,\mathrm{ir}}+\beta_{m,\mathrm{vis}})h_m}}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m1,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,s,y}\chi_{y\beta\gamma}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5.24b}
$$

当背面 和频极化由薄膜 ($m''$) 引起时：

$$
\begin{aligned}
E_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}t_{mm^{\prime\prime},\mathrm{vis},\beta}(1+r_{m^{\prime\prime}2,\mathrm{vis},\beta})e^{i\beta_{m,\mathrm{vis}}h_m}}{(1+r_{mm^{\prime\prime},\mathrm{vis},\beta}r_{m^{\prime\prime}2,\mathrm{vis},\beta})+r_{1m,\mathrm{vis},\beta}(r_{mm^{\prime\prime},\mathrm{vis},\beta}+r_{m^{\prime\prime}2,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}t_{mm^{\prime\prime},\mathrm{ir},\gamma}(1+r_{m^{\prime\prime}2,\mathrm{ir},\gamma})e^{i\beta_{m,\mathrm{ir}}h_m}}{(1+r_{mm^{\prime\prime},\mathrm{ir},\gamma}r_{m^{\prime\prime}2,\mathrm{ir},\gamma})+r_{1m,\mathrm{ir},\gamma}(r_{mm^{\prime\prime},\mathrm{ir},\gamma}+r_{m^{\prime\prime}2,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{(1+r_{mm^{\prime\prime},\mathrm{SF},p}r_{m^{\prime\prime}2,\mathrm{SF},p})[(1-r_{1m,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,p,x}\chi_{x\beta\gamma}+(1+r_{1m,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,p,z}\chi_{z\beta\gamma}]}{1+r_{mm^{\prime\prime},\mathrm{SF},p}r_{m^{\prime\prime}2,\mathrm{SF},p}+r_{1m,\mathrm{SF},p}(r_{mm^{\prime\prime},\mathrm{SF},p}+r_{m^{\prime\prime}2,\mathrm{SF},p})e^{2i\beta_{m,\mathrm{SF}}h_m}}
\end{aligned}\tag{5.25a}
$$

$$
\begin{aligned}
E_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}t_{mm^{\prime\prime},\mathrm{vis},\beta}(1+r_{m^{\prime\prime}2,\mathrm{vis},\beta})e^{i\beta_{m,\mathrm{vis}}h_m}}{(1+r_{mm^{\prime\prime},\mathrm{vis},\beta}r_{m^{\prime\prime}2,\mathrm{vis},\beta})+r_{1m,\mathrm{vis},\beta}(r_{mm^{\prime\prime},\mathrm{vis},\beta}+r_{m^{\prime\prime}2,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}t_{mm^{\prime\prime},\mathrm{ir},\gamma}(1+r_{m^{\prime\prime}2,\mathrm{ir},\gamma})e^{i\beta_{m,\mathrm{ir}}h_m}}{(1+r_{mm^{\prime\prime},\mathrm{ir},\gamma}r_{m^{\prime\prime}2,\mathrm{ir},\gamma})+r_{1m,\mathrm{ir},\gamma}(r_{mm^{\prime\prime},\mathrm{ir},\gamma}+r_{m^{\prime\prime}2,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{(1+r_{mm^{\prime\prime},\mathrm{SF},s}r_{m^{\prime\prime}2,\mathrm{SF},s})[(1-r_{1m,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{+}_{2/m,s,y}\chi_{y\beta\gamma}]}{1+r_{mm^{\prime\prime},\mathrm{SF},s}r_{m^{\prime\prime}2,\mathrm{SF},s}+r_{1m,\mathrm{SF},s}(r_{mm^{\prime\prime},\mathrm{SF},s}+r_{m^{\prime\prime}2,\mathrm{SF},s})e^{2i\beta_{m,\mathrm{SF}}h_m}}
\end{aligned}\tag{5.25b}
$$
