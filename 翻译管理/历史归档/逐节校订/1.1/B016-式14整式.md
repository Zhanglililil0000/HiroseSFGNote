# 式（14）整式

$$
\begin{aligned}
\chi^{(3)}_{ijkl}(\omega_0=\omega_1+\omega_2+\omega_3)
&=\frac{P_i^{(3)}(\omega_0=\omega_1+\omega_2+\omega_3)}{E_j(\omega_1)E_k(\omega_2)E_l(\omega_3)}\\[8pt]
&=-N\frac{e^4}{\hbar^3}\sum_{g,n,n\prime,n\prime\prime}\Biggl\{\\[8pt]
&\quad \frac{(r_l)_{ng}}{\omega_3 - \omega_{ng} + i\Gamma_{ng}}\\[6pt]
&\qquad \times\Biggl[\frac{(r_j)_{n' n}}{\omega_1 + \omega_3 - \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{g n''} (r_k)_{n'' n'}}{\omega_0 - \omega_{n''g} + i\Gamma_{n''g}} - \frac{(r_k)_{g n''} (r_i)_{n'' n'}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{n' n}}{\omega_2 + \omega_3 - \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{g n''} (r_j)_{n'' n'}}{\omega_0 - \omega_{n''g} + i\Gamma_{n''g}} - \frac{(r_j)_{g n''} (r_i)_{n'' n'}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_j)_{g n''}}{\omega_1 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_k)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_k)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{g n''}}{\omega_2 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_j)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_j)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\Biggr]\\[8pt]
&\quad +\frac{(r_l)_{g n''}}{\omega_3 + \omega_{n''g} + i\Gamma_{n''g}}\\[6pt]
&\qquad \times\Biggl[\frac{(r_j)_{n'' n'}}{\omega_1 + \omega_3 + \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{n' n} (r_k)_{n g}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_k)_{n' n} (r_i)_{n g}}{\omega_0 + \omega_{ng} + i\Gamma_{ng}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{n'' n'}}{\omega_2 + \omega_3 + \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{n' n} (r_j)_{n g}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_j)_{n' n} (r_i)_{n g}}{\omega_0 + \omega_{ng} + i\Gamma_{ng}}\right]\\[8pt]
&\qquad +\frac{(r_j)_{n g}}{\omega_1 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_k)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_k)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{n g}}{\omega_2 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_j)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_j)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\Biggr]\\[8pt]
&\quad +\frac{(r_k)_{n g}}{\omega_2 - \omega_{ng} + i\Gamma_{ng}}\\[6pt]
&\qquad \times\Biggl[\frac{(r_l)_{n' n}}{\omega_2 + \omega_3 - \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{g n''} (r_j)_{n'' n'}}{\omega_0 - \omega_{n''g} + i\Gamma_{n''g}} - \frac{(r_j)_{g n''} (r_i)_{n'' n'}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_l)_{g n''}}{\omega_2 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_j)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_j)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_j)_{n' n}}{\omega_1 + \omega_2 - \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{g n''} (r_l)_{n'' n'}}{\omega_0 - \omega_{n''g} + i\Gamma_{n''g}} - \frac{(r_l)_{g n''} (r_i)_{n'' n'}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_j)_{g n''}}{\omega_1 + \omega_2 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_l)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_l)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\Biggr]\\[8pt]
&\quad +\frac{(r_k)_{g n''}}{\omega_2 + \omega_{n''g} + i\Gamma_{n''g}}\\[6pt]
&\qquad \times\Biggl[\frac{(r_l)_{n'' n'}}{\omega_2 + \omega_3 + \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{n' n} (r_j)_{n g}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_j)_{n' n} (r_i)_{n g}}{\omega_0 + \omega_{ng} + i\Gamma_{ng}}\right]\\[8pt]
&\qquad +\frac{(r_l)_{n g}}{\omega_2 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_j)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_j)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_j)_{n'' n'}}{\omega_1 + \omega_2 + \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{n' n} (r_l)_{n g}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_l)_{n' n} (r_i)_{n g}}{\omega_0 + \omega_{ng} + i\Gamma_{ng}}\right]\\[8pt]
&\qquad +\frac{(r_j)_{n g}}{\omega_1 + \omega_2 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_l)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_l)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\Biggr]\\[8pt]
&\quad +\frac{(r_j)_{n g}}{\omega_1 - \omega_{ng} + i\Gamma_{ng}}\\[6pt]
&\qquad \times\Biggl[\frac{(r_l)_{n' n}}{\omega_1 + \omega_3 - \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{g n''} (r_k)_{n'' n'}}{\omega_0 - \omega_{n''g} + i\Gamma_{n''g}} - \frac{(r_k)_{g n''} (r_i)_{n'' n'}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_l)_{g n''}}{\omega_1 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_k)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_k)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{n' n}}{\omega_1 + \omega_2 - \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{g n''} (r_l)_{n'' n'}}{\omega_0 - \omega_{n''g} + i\Gamma_{n''g}} - \frac{(r_l)_{g n''} (r_i)_{n'' n'}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{g n''}}{\omega_1 + \omega_2 - \omega_{n n''} + i\Gamma_{n n''}} \left[ \frac{(r_l)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_l)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\Biggr]\\[8pt]
&\quad +\frac{(r_j)_{g n''}}{\omega_1 + \omega_{n''g} + i\Gamma_{n''g}}\\[6pt]
&\qquad \times\Biggl[\frac{(r_l)_{n'' n'}}{\omega_1 + \omega_3 + \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{n' n} (r_k)_{n g}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_k)_{n' n} (r_i)_{n g}}{\omega_0 + \omega_{ng} + i\Gamma_{ng}}\right]\\[8pt]
&\qquad +\frac{(r_l)_{n g}}{\omega_1 + \omega_3 - \omega_{n n''} + i\Gamma_{n n''}} \left[\frac{(r_k)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_k)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{n'' n'}}{\omega_1 + \omega_2 + \omega_{n'g} + i\Gamma_{n'g}} \left[ \frac{(r_i)_{n' n} (r_l)_{n g}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_l)_{n' n} (r_i)_{n g}}{\omega_0 + \omega_{ng} + i\Gamma_{ng}}\right]\\[8pt]
&\qquad +\frac{(r_k)_{n g}}{\omega_1 + \omega_2 - \omega_{n n''} + i\Gamma_{n n''}} \left[\frac{(r_l)_{n'' n'} (r_i)_{n' n}}{\omega_0 - \omega_{n n'} + i\Gamma_{n n'}} - \frac{(r_i)_{n'' n'} (r_l)_{n' n}}{\omega_0 - \omega_{n' n''} + i\Gamma_{n' n''}}\right]\Biggr]\\[8pt]
&\Biggr\}
\end{aligned}
\tag{14}
$$

> **排版说明：** 原文部分加项另加方括号，本式统一按“六主组、每组四子项”的层次排写；各子项内的差项括号保留，加项次序及符号不变。
