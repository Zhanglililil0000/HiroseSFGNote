# B025：附录C分批校订



上述方程的解为：

$$
\begin{aligned}
&\rho^{(2)}(\omega_1 + \omega_2)_{nn'} = \frac{e^2}{\hbar^2} E_j(\omega_1) E_k(\omega_2) \frac{1}{\omega_1 + \omega_2 - \omega_{nn'} + i\Gamma_{nn'}}\\[6pt]
&\quad \times \Bigg\{ \sum_{n''} \frac{(r_j)_{nn''} (r_k)_{n''n'}}{\omega_2 - \omega_{n''n'} + i\Gamma_{n''n'}} (\rho^{(0)}_{n'n'} - \rho^{(0)}_{n''n''})\\[6pt]
&\quad - \frac{(r_k)_{nn''} (r_j)_{n''n'}}{\omega_2 - \omega_{nn''} + i\Gamma_{nn''}} (\rho^{(0)}_{n''n''} - \rho^{(0)}_{nn})\\[6pt]
&\quad + \frac{(r_k)_{nn''} (r_j)_{n''n'}}{\omega_1 - \omega_{n''n'} + i\Gamma_{n''n'}} (\rho^{(0)}_{n'n'} - \rho^{(0)}_{n''n''})\\[6pt]
&\quad - \frac{(r_k)_{nn''} (r_j)_{n''n'}}{\omega_1 - \omega_{nn''} + i\Gamma_{nn''}} (\rho^{(0)}_{n''n''} - \rho^{(0)}_{nn}) \Bigg\}
\end{aligned}
\tag{C-6}
$$

$$
\begin{aligned}
&\rho^{(2)}(\omega_1 - \omega_2)_{nn'} = \frac{e^2}{\hbar^2} E_j(\omega_1) E_k(-\omega_2) \frac{1}{\omega_1 - \omega_2 - \omega_{nn'} + i\Gamma_{nn'}}\\[6pt]
&\quad \times \Bigg\{ - \sum_{n''} \frac{(r_j)_{nn''} (r_k)_{n''n'}}{\omega_2 + \omega_{n''n'} - i\Gamma_{n''n'}} (\rho^{(0)}_{n'n'} - \rho^{(0)}_{n''n''})\\[6pt]
&\quad + \frac{(r_k)_{nn''} (r_j)_{n''n'}}{\omega_2 + \omega_{nn''} - i\Gamma_{nn''}} (\rho^{(0)}_{n''n''} - \rho^{(0)}_{nn})\\[6pt]
&\quad + \frac{(r_k)_{nn''} (r_j)_{n''n'}}{\omega_1 - \omega_{n''n'} + i\Gamma_{n''n'}} (\rho^{(0)}_{n'n'} - \rho^{(0)}_{n''n''})\\[6pt]
&\quad - \frac{(r_j)_{nn''} (r_k)_{n''n'}}{\omega_1 - \omega_{nn''} + i\Gamma_{nn''}} (\rho^{(0)}_{n''n''} - \rho^{(0)}_{nn}) \Bigg\}
\end{aligned}
\tag{C-7}
$$

$$
\begin{aligned}
&\rho^{(2)}(2\omega)_{nn'} = \frac{e^2}{\hbar^2} E_j(\omega) E_k(\omega) \frac{1}{2\omega - \omega_{nn'} + i\Gamma_{nn'}}\\[6pt]
&\quad \times \Bigg[ \sum_{n''} \frac{(r_k)_{nn''} (r_j)_{n''n'} + (r_j)_{nn''} (r_k)_{n''n'}}{\omega_1 - \omega_{n''n'} + i\Gamma_{n''n'}} (\rho^{(0)}_{n'n'} - \rho^{(0)}_{n''n''})\\[6pt]
&\quad - \frac{(r_k)_{nn''} (r_j)_{n''n'} + (r_j)_{nn''} (r_k)_{n''n'}}{\omega_1 - \omega_{nn''} + i\Gamma_{nn''}} (\rho^{(0)}_{n''n''} - \rho^{(0)}_{nn}) \Bigg]
\end{aligned}
\tag{C-8}
$$

$$
\begin{aligned}
&\rho^{(2)}(0)_{nn'} = \frac{e^2}{\hbar^2} E_j(\omega) E_k(-\omega) \frac{1}{-\omega_{nn'} + i\Gamma_{nn'}}\\[6pt]
&\quad \times \Bigg\{ \bigl[ \sum_{n''} \frac{(r_k)_{nn''} (r_j)_{n''n'}}{\omega - \omega_{n''n'} + i\Gamma_{n''n'}} - \frac{(r_j)_{nn''} (r_k)_{n''n'}}{\omega + \omega_{n''n'} - i\Gamma_{n''n'}} \bigr] (\rho^{(0)}_{n'n'}\\[6pt]
&- \rho^{(0)}_{n''n''})\\[6pt]
&\quad - \bigl[ \frac{(r_j)_{nn''} (r_k)_{n''n'}}{\omega - \omega_{nn''} + i\Gamma_{nn''}} - \frac{(r_k)_{nn''} (r_j)_{n''n'}}{\omega + \omega_{nn''} - i\Gamma_{nn''}} \bigr] (\rho^{(0)}_{n''n''} - \rho^{(0)}_{nn}) \Bigg\}
\end{aligned}
\tag{C-9}
$$

在下面的通用公式中，假设（最后引入的）相互作用项中考虑的光的频率为 $\omega_j$，其电场的方向为 $j$ 轴，则以下关系式成立。

$$
\begin{aligned}
&\frac{\partial \rho^{(l)}(\omega)_{nn'}}{\partial t} = -i \bigl[ \omega_{nn'} \rho^{(l)}(\omega)_{nn'} - i\Gamma_{nn'} \rho^{(l)}(\omega)_{nn'} + \frac{(H'_{\text{int}}, \rho^{(l-1)}(\omega - \omega_j)_{nn'})}{\hbar} \bigr]\\[6pt]
&\rho^{(l)}(\omega)_{nn'} = \frac{e}{\hbar} E_j(\omega_j) \frac{\sum_{n''} \bigl[ (r_j)_{nn''} \rho^{(l-1)}(\omega - \omega_j)_{n''n'} - (r_j)_{n''n'} \rho^{(l-1)}(\omega - \omega_j)_{nn''} \bigr]}{\omega - \omega_{nn'} + i\Gamma_{nn'}}
\end{aligned}
\tag{C-10}
$$
