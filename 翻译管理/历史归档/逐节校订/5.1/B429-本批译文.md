### 式(5c)：首指标为z的分量

$$
\begin{aligned}
\beta_{zxx}={}&\frac{Y^{(1)}_{0,1}}{\sqrt{8}}+\frac{Y^{(1)}_{0,2}}{2}-\frac{Y^{(1)}_{0,3}}{\sqrt{40}}\\
&+\frac{Y^{(2)}_{2a,1}}{2}-\frac{Y^{(2)}_{2a,2}}{\sqrt{12}}+\frac{Y^{(3)}_{0}}{\sqrt{10}}\\
&-\frac{Y^{(3)}_{2b}}{\sqrt{6}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zyy}={}&\frac{Y^{(1)}_{0,1}}{\sqrt{8}}+\frac{Y^{(1)}_{0,2}}{2}-\frac{Y^{(1)}_{0,3}}{\sqrt{40}}\\
&-\frac{Y^{(2)}_{2a,1}}{2}+\frac{Y^{(2)}_{2a,2}}{\sqrt{12}}+\frac{Y^{(3)}_{0}}{\sqrt{10}}\\
&+\frac{Y^{(3)}_{2b}}{\sqrt{6}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zzz}={}&\frac{2Y^{(1)}_{0,1}}{\sqrt{8}}+\frac{2Y^{(1)}_{0,3}}{\sqrt{40}}-\frac{2Y^{(3)}_{0}}{\sqrt{10}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zxy}={}&\frac{Y^{(0)}_{0}}{\sqrt{6}}+\frac{Y^{(2)}_{0,1}}{\sqrt{12}}+\frac{Y^{(2)}_{0,2}}{2}\\
&-\frac{Y^{(2)}_{02b,1}}{2}+\frac{Y^{(2)}_{2b,2}}{\sqrt{12}}+\frac{Y^{(3)}_{2a}}{\sqrt{6}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zyx}={}&-\frac{Y^{(0)}_{0}}{\sqrt{6}}-\frac{Y^{(2)}_{0,1}}{\sqrt{12}}-\frac{Y^{(2)}_{0,2}}{2}\\
&-\frac{Y^{(2)}_{02b,1}}{2}+\frac{Y^{(2)}_{2b,2}}{\sqrt{12}}+\frac{Y^{(3)}_{2a}}{\sqrt{6}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zyz}={}&\frac{4Y^{(1)}_{1a,3}}{\sqrt{40}}+\frac{2Y^{(2)}_{1b,2}}{\sqrt{12}}+\frac{4Y^{(3)}_{1a}}{\sqrt{60}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zzy}={}&\frac{Y^{(1)}_{1a,1}}{\sqrt{8}}-\frac{Y^{(1)}_{1a,2}}{2}-\frac{Y^{(1)}_{1a,3}}{\sqrt{40}}\\
&-\frac{Y^{(2)}_{1b,1}}{2}-\frac{Y^{(2)}_{1b,2}}{\sqrt{12}}+\frac{4Y^{(3)}_{1a}}{\sqrt{60}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zzx}={}&\frac{Y^{(1)}_{1b,1}}{\sqrt{8}}-\frac{Y^{(1)}_{1b,2}}{2}-\frac{Y^{(1)}_{1b,3}}{\sqrt{40}}\\
&+\frac{Y^{(2)}_{1a,1}}{2}+\frac{Y^{(2)}_{1a,2}}{\sqrt{12}}+\frac{4Y^{(3)}_{1b}}{\sqrt{60}}
\end{aligned}
$$

$$
\begin{aligned}
\beta_{zxz}={}&\frac{4Y^{(1)}_{1b,3}}{\sqrt{40}}-\frac{2Y^{(2)}_{1a,2}}{\sqrt{12}}+\frac{4Y^{(3)}_{1b}}{\sqrt{60}}
\end{aligned}\tag{5c}
$$
