## 1/m界面外侧产生的和频光

源点为 $x=-2nh_m\tan\theta_{m,\mathrm{SF}}$、$z=0^-$。

### 反射和频光


$$
\begin{aligned}
E_{1p}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,x}\chi_{x\beta\gamma}+(1-r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,z}\chi_{z\beta\gamma}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{1a}
$$


$$
\begin{aligned}
E_{1s}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,s,y}\chi_{y\beta\gamma}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{1b}
$$

### 透射和频光


$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{SF},p}t_{m2,\mathrm{SF},p}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}(L^{+}_{m/1,p,x}\chi_{x\beta\gamma}+L^{+}_{m/1,p,z}\chi_{z\beta\gamma})
\end{aligned}\tag{2a}
$$


$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{(1+r_{1m,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{(1+r_{1m,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{SF},s}t_{m2,\mathrm{SF},s}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}L^{+}_{m/1,s,y}\chi_{y\beta\gamma}
\end{aligned}\tag{2b}
$$

> **原文疑点（Q080，PDF p4，式2a、2b）**：这里的L层标原为 $m/1$，而定义段用 $1/m$。此处保留；是否应统一为 $1/m$ 待核。式1a、1b的场振幅原未写方向上标，也不补入。

### 表面已形成完整薄膜时

当表面已形成完整薄膜时，构成该薄膜（原文用 $m'$ 标示其折射率）的分子所产生的和频光需要考虑内部反射。

#### 反射和频光


$$
\begin{aligned}
E^-_{1p}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{(1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p})[(1+r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,x}\chi_{x\beta\gamma}+(1-r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,z}\chi_{z\beta\gamma}]}{1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p}+r_{m2,\mathrm{SF},p}(r_{1m',\mathrm{SF},p}+r_{m'm,\mathrm{SF},p})e^{2i\beta_{m,\mathrm{SF}}h_m}}
\end{aligned}\tag{3a}
$$


$$
\begin{aligned}
E^-_{1s}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{(1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s})[(1+r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,s,y}\chi_{y\beta\gamma}]}{1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s}+r_{m2,\mathrm{SF},s}(r_{1m',\mathrm{SF},s}+r_{m'm,\mathrm{SF},s})e^{2i\beta_{m,\mathrm{SF}}h_m}}
\end{aligned}\tag{3b}
$$

#### 透射和频光


$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{t_{m2,\mathrm{SF},p}(1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p})e^{i\beta_{m,\mathrm{SF}}h_m}}{1+r_{1m',\mathrm{SF},p}r_{m'm,\mathrm{SF},p}+r_{m2,\mathrm{SF},p}(r_{mm',\mathrm{SF},p}+r_{m'1,\mathrm{SF},p})e^{2i\beta_{m,\mathrm{SF}}h_m}}\mathclose{)}(L^{+}_{1/m,p,x}\chi_{x\beta\gamma}+L^{+}_{1/m,p,z}\chi_{z\beta\gamma})
\end{aligned}\tag{4a}
$$


$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m',\mathrm{vis},\beta}(1+r_{m'm,\mathrm{vis},\beta})(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{1+r_{1m',\mathrm{vis},\beta}r_{m'm,\mathrm{vis},\beta}+r_{m2,\mathrm{vis},\beta}(r_{1m',\mathrm{vis},\beta}+r_{m'm,\mathrm{vis},\beta})e^{2i\beta_{m,\mathrm{vis}}h_m}} \\
&\quad\times \frac{t_{1m',\mathrm{ir},\gamma}(1+r_{m'm,\mathrm{ir},\gamma})(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{1+r_{1m',\mathrm{ir},\gamma}r_{m'm,\mathrm{ir},\gamma}+r_{m2,\mathrm{ir},\gamma}(r_{1m',\mathrm{ir},\gamma}+r_{m'm,\mathrm{ir},\gamma})e^{2i\beta_{m,\mathrm{ir}}h_m}} \\
&\quad\times \frac{t_{m2,\mathrm{SF},s}(1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s})e^{i\beta_{m,\mathrm{SF}}h_m}}{1+r_{1m',\mathrm{SF},s}r_{m'm,\mathrm{SF},s}+r_{m2,\mathrm{SF},s}(r_{mm',\mathrm{SF},s}+r_{m'1,\mathrm{SF},s})e^{2i\beta_{m,\mathrm{SF}}h_m}}\mathclose{)}L^{+}_{1/m,s,y}\chi_{y\beta\gamma}
\end{aligned}\tag{4b}
$$

> **原文疑点（Q081，PDF p5，式4a、4b）**：和频出射分母的末项原用 $r_{mm',\mathrm{SF},p(s)}+r_{m'1,\mathrm{SF},p(s)}$，与式3的方向下标不同，且分式之后有一个未配对的右圆括号。两处均保留；方向改用式3及删去该括号只是候选，未采用。
