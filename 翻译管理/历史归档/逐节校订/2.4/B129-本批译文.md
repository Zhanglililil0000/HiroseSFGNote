## 1/m界面内侧产生的和频光

源点为 $x=-2nh_m\tan\theta_{m,\mathrm{SF}}$、$z=0^+$。

### 反射和频光


$$
\begin{aligned}
E^-_{1p}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,x}\chi_{x\beta\gamma}+(1-r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,p,z}\chi_{z\beta\gamma}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5a}
$$


$$
\begin{aligned}
E^-_{1s}(0^-)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{(1+r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})L^{-}_{1/m,s,y}\chi_{y\beta\gamma}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}
\end{aligned}\tag{5b}
$$

### 透射和频光


$$
\begin{aligned}
E^+_{2p}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{m2,\mathrm{SF},p}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},p}r_{m2,\mathrm{SF},p}e^{2i\beta_{m,\mathrm{SF}}h_m})}(L^{+}_{1/m,p,x}\chi_{x\beta\gamma}+L^{+}_{1/m,p,z}\chi_{z\beta\gamma})
\end{aligned}\tag{6a}
$$


$$
\begin{aligned}
E^+_{2s}(h_m^+)&=\sum_{\beta,\gamma}E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma} \frac{t_{1m,\mathrm{vis},\beta}(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})}{(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m})} \\
&\quad\times \frac{t_{1m,\mathrm{ir},\gamma}(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})}{(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m})} \\
&\quad\times \frac{t_{m2,\mathrm{SF},s}e^{i\beta_{m,\mathrm{SF}}h_m}}{(1+r_{1m,\mathrm{SF},s}r_{m2,\mathrm{SF},s}e^{2i\beta_{m,\mathrm{SF}}h_m})}L^{+}_{1/m,s,y}\chi_{y\beta\gamma}
\end{aligned}\tag{6b}
$$
