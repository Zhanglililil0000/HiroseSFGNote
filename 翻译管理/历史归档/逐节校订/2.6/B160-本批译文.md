### 4.3. 三层薄膜（1/m'/m/m''/2）中的薄膜内电场

设薄膜厚度从入射侧开始依次为 $h_{m'}$、$h_m$、$h_{m''}$，则有：

$$
\begin{aligned}
\begin{bmatrix}E^+(0)\\[6pt]
E^-(0)\end{bmatrix}
&= \frac{1}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}\\[6pt]
{\begin{aligned}&(r_{1m'} + r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{1m'} r_{m'm} + e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}\end{bmatrix}
\end{aligned}\tag{4.15a}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_1)\\[6pt]
E^-(z_1)\end{bmatrix}
&= \frac{t_{1m'} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}[(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}}) + r_{m'm}(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}] e^{i\beta_{m'} z_1}\\[6pt]
[r_{m'm}(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}}) + (r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}] e^{2i\beta_{m'} h_{m'}} e^{-i\beta_{m'} z_1}\end{bmatrix}
\end{aligned}\tag{4.16a}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_2)\\[6pt]
E^-(z_2)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} e^{i\beta_{m'} h_{m'}} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{i\beta_m z_2}\\[6pt]
(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m} e^{-i\beta_m z_2}\end{bmatrix}
\end{aligned}\tag{4.17a}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_3)\\[6pt]
E^-(z_3)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{mm'''} e^{i(\beta_{m'} h_{m'} + \beta_m h_m)} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm'''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}e^{i\beta_{m''} z_3}\\[6pt]
r_{m''2} e^{2i\beta_{m''} h_{m''}} e^{-i\beta_{m''} z_3}\end{bmatrix}
\end{aligned}\tag{4.18a}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_4)\\[6pt]
E^-(z_4)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{mm''} t_{m''2} e^{i(\beta_{m'} h_{m'} + \beta_m h_m + \beta_{m''} h_{m''})} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}e^{i\beta_{2'} z_4}\\[6pt]
0\end{bmatrix}
\end{aligned}\tag{4.19a}
$$

因此，

$$
\begin{aligned}
\begin{bmatrix}E^+(0) + E^-(0)\\[6pt]
E^+(0) - E^-(0)\end{bmatrix}
&= \frac{1}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}(1 + r_{1m'})[{\begin{aligned}&(1 + r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}]\\[6pt]
(1 - r_{1m'})[{\begin{aligned}&(1 - r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} - e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}]\end{bmatrix}
\end{aligned}\tag{4.15b}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_1) + E^-(z_1)\\[6pt]
E^+(z_1) - E^-(z_1)\end{bmatrix}
&= \frac{t_{1m'} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}{\begin{aligned}&(e^{i\beta_{m'} z_1} + r_{m'm} e^{i\beta_{m'}(2h_{m'} - z_1)})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} e^{i\beta_{m'} z_1} + e^{i\beta_{m'}(2h_{m'} - z_1)})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}\\[6pt]
{\begin{aligned}&(e^{i\beta_{m'} z_1} - r_{m'm} e^{i\beta_{m'}(2h_{m'} - z_1)})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} e^{i\beta_{m'} z_1} - e^{i\beta_{m'}(2h_{m'} - z_1)})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}\end{bmatrix}
\end{aligned}\tag{4.16b}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_2) + E^-(z_2)\\[6pt]
E^+(z_2) - E^-(z_2)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} e^{i\beta_{m'} h_{m'}} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{i\beta_m z_2} + (r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m} e^{-i\beta_m z_2}\\[6pt]
(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{i\beta_m z_2} - (r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m} e^{-i\beta_m z_2}\end{bmatrix}
\end{aligned}\tag{4.17b}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_3) + E^-(z_3)\\[6pt]
E^+(z_3) - E^-(z_3)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{mm''} e^{i(\beta_{m'} h_{m'} + \beta_m h_m)} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}e^{i\beta_{m''} z_3} + r_{m''2} e^{2i\beta_{m''} h_{m''}} e^{-i\beta_{m''} z_3}\\[6pt]
e^{i\beta_{m''} z_3} - r_{m''2} e^{2i\beta_{m''} h_{m''}} e^{-i\beta_{m''} z_3}\end{bmatrix}
\end{aligned}\tag{4.18b}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_4) + E^-(z_4)\\[6pt]
E^+(z_4) - E^-(z_4)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{mm''} t_{m''2} e^{i(\beta_{m'} h_{m'} + \beta_m h_m + \beta_{m''} h_{m''})} E^+(0)}{{\begin{aligned}&(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}})(1 + r_{mm''} r_{m''2} e^{2i\beta_{m''} h_{m''}})\\&+ (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}})(r_{mm''} + r_{m''2} e^{2i\beta_{m''} h_{m''}}) e^{2i\beta_m h_m}\end{aligned}}}\\
&\quad\times \begin{bmatrix}e^{i\beta_{2'} z_4} + e^{-i\beta_2 z_4}\\[6pt]
e^{i\beta_{2'} z_4} - e^{-i\beta_2 z_4}\end{bmatrix}
\end{aligned}\tag{4.19b}
$$



> **原文疑点（Q092，PDF p9—11）**：4.18a局部出现 $t_{mm^{\prime\prime\prime}}$ 及分母 $r_{mm^{\prime\prime\prime}}$，4.19a/b的前行相位写 $\beta_{2\prime}$，保留额外撇号，候选为双撇号层标或无撇号2。4.19b在已经规定末介质无−波后仍有 $e^{-i\beta_2z_4}$ 项，未直接删去。



> **原文疑点（Q093，PDF p9—10，式4.15a/b）**：右侧前因子原为1，没有像邻式那样写 $E^+(0)$；与左端场幅的归一化关系不明。此处照录，不补入场幅。候选为作者在此归一化入射场或漏写该因子。
