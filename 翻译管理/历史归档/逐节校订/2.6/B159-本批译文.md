**表面一侧薄膜厚度为零的极限（$h_{m'} \to 0$）下：**

$$
\begin{aligned}
\begin{bmatrix}E^+(0) + E^-(0)\\[6pt]
E^+(0) - E^-(0)\end{bmatrix}
&= \frac{E^+(0)}{(1 + r_{1m'} r_{m'm}) + (r_{m'm} + r_{1m'}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}(1 + r_{1m'})(1 + r_{m'm})(1 + r_{m2} e^{2i\beta_m h_m})\\[6pt]
(1 - r_{1m'})(1 - r_{m'm})(1 - r_{m2} e^{2i\beta_m h_m})\end{bmatrix} \tag{4.11a}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_1) + E^-(z_1)\\[6pt]
E^+(z_1) - E^-(z_1)\end{bmatrix}
&= \frac{t_{1m'} E^+(0)}{(1 + r_{1m'} r_{m'm}) + (r_{m'm} + r_{1m'}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}(1 + r_{m'm})(1 + r_{m2} e^{2i\beta_m h_m})\\[6pt]
(1 - r_{m'm})(1 - r_{m2} e^{2i\beta_m h_m})\end{bmatrix} \tag{4.12a}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_2) + E^-(z_2)\\[6pt]
E^+(z_2) - E^-(z_2)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} E^+(0)}{(1 + r_{1m'} r_{m'm}) + (r_{m'm} + r_{1m'}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}e^{i\beta_m z_2} + r_{m2} e^{2i\beta_m h_m} e^{-i\beta_m z_2}\\[6pt]
e^{i\beta_m z_2} - r_{m2} e^{2i\beta_m h_m} e^{-i\beta_m z_2}\end{bmatrix} \tag{4.13a}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_3) + E^-(z_3)\\[6pt]
E^+(z_3) - E^-(z_3)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{m2} e^{i\beta_m h_m} E^+(0)}{(1 + r_{1m'} r_{m'm}) + (r_{m'm} + r_{1m'}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}e^{i\beta_2 z_3}\\[6pt]
e^{i\beta_2 z_3}\end{bmatrix} \tag{4.14a}
\end{aligned}
$$

**背面一侧薄膜厚度为零的极限（$h_m \to 0$）下：**

$$
\begin{aligned}
\begin{bmatrix}E^+(0) + E^-(0)\\[6pt]
E^+(0) - E^-(0)\end{bmatrix}
&= \frac{E^+(0)}{(1 + r_{m'm} r_{m2}) + r_{1m'}(r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}}}
\begin{bmatrix}(1 + r_{1m'})[(1 + r_{m'm} r_{m2}) + (r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}}]\\[6pt]
(1 - r_{1m'})[(1 + r_{m'm} r_{m2}) - (r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}}]\end{bmatrix} \tag{4.11b}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_1) + E^-(z_1)\\[6pt]
E^+(z_1) - E^-(z_1)\end{bmatrix}
&= \frac{t_{1m'} E^+(0)}{(1 + r_{m'm} r_{m2}) + r_{1m'}(r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}}}
\begin{bmatrix}(1 + r_{m'm} r_{m2}) e^{i\beta_{m'} z_1} + (r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}} e^{-i\beta_{m'} z_1}\\[6pt]
(1 + r_{m'm} r_{m2}) e^{i\beta_{m'} z_1} - (r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}} e^{-i\beta_{m'} z_1}\end{bmatrix} \tag{4.12b}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_2) + E^-(z_2)\\[6pt]
E^+(z_2) - E^-(z_2)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} e^{i\beta_{m'} h_{m'}} E^+(0)}{(1 + r_{m'm} r_{m2}) + r_{1m'}(r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}}}
\begin{bmatrix}1 + r_{m2}\\[6pt]
1 - r_{m2}\end{bmatrix} \tag{4.13b}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_3) + E^-(z_3)\\[6pt]
E^+(z_3) - E^-(z_3)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{m2} e^{i\beta_{m'} h_{m'}} E^+(0)}{(1 + r_{m'm} r_{m2}) + r_{1m'}(r_{m'm} + r_{m2}) e^{2i\beta_{m'} h_{m'}}}
\begin{bmatrix}e^{i\beta_2 z_3}\\[6pt]
e^{i\beta_2 z_3}\end{bmatrix} \tag{4.14b}
\end{aligned}
$$
