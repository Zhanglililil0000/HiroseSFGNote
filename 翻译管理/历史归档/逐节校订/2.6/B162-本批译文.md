**表面一侧和背面一侧薄膜厚度均为零时（$h_{m'}, h_{m''} \to 0$）：**

$$
\begin{aligned}
\begin{bmatrix}E^+(0)\\[6pt]
E^-(0)\end{bmatrix}
&= \frac{1}{(1 + r_{1m'} r_{m'm})(1 + r_{mm''} r_{m''2}) + (r_{1m'} + r_{m'm})(r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}}\\
&\quad\times \begin{bmatrix}(1 + r_{1m'} r_{m'm})(1 + r_{mm''} r_{m''2}) + (r_{1m'} + r_{m'm})(r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}\\[6pt]
(r_{1m'} + r_{m'm})(1 + r_{mm''} r_{m''2}) + (1 + r_{1m'} r_{m'm})(r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}\end{bmatrix}
\tag{4.25a}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(0) + E^-(0)\\[6pt]
E^+(0) - E^-(0)\end{bmatrix}
&= \frac{1}{(1 + r_{1m'} r_{m'm})(1 + r_{mm''} r_{m''2}) + (r_{1m'} + r_{m'm})(r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}}\\
&\quad\times \begin{bmatrix}\mathopen{[}(1 + r_{1m'})(1 + r_{m'm})[(1 + r_{mm''} r_{m''2}) + (r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}]\\[6pt]
\mathopen{[}(1 - r_{1m'})(1 - r_{m'm})[(1 + r_{mm''} r_{m''2}) - (r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}]\end{bmatrix}
\tag{4.25b}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_1) + E^-(z_1)\\[6pt]
E^+(z_1) - E^-(z_1)\end{bmatrix}
&= \frac{t_{1m'} E^+(0)}{(1 + r_{1m'} r_{m'm})(1 + r_{mm''} r_{m''2}) + (r_{1m'} + r_{m'm})(r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}}\\
&\quad\times \begin{bmatrix}(1 + r_{m'm})[(1 + r_{mm''} r_{m''2}) + (r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}]\\[6pt]
(1 - r_{m'm})[(1 + r_{mm''} r_{m''2}) - (r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}]\end{bmatrix}
\tag{4.26}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_2) + E^-(z_2)\\[6pt]
E^+(z_2) - E^-(z_2)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} E^+(0)}{(1 + r_{1m'} r_{m'm})(1 + r_{mm''} r_{m''2}) + (r_{1m'} + r_{m'm})(r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}}\\
&\quad\times \begin{bmatrix}(1 + r_{mm''} r_{m''2}) e^{i\beta_m z_2} + (r_{mm''} + r_{m''2}) e^{2i\beta_m h_m} e^{-i\beta_m z_2}\\[6pt]
(1 + r_{mm''} r_{m''2}) e^{i\beta_m z_2} - (r_{mm''} + r_{m''2}) e^{2i\beta_m h_m} e^{-i\beta_m z_2}\end{bmatrix}
\tag{4.27}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_3) + E^-(z_3)\\[6pt]
E^+(z_3) - E^-(z_3)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{mm''} e^{i\beta_m h_m} E^+(0)}{(1 + r_{1m'} r_{m'm})(1 + r_{mm''} r_{m''2}) + (r_{1m'} + r_{m'm})(r_{mm''} + r_{m''2}) e^{2i\beta_m h_m}}\\
&\quad\times \begin{bmatrix}e^{i\beta_{m''} z_3} + r_{m''2} e^{-i\beta_{m''} z_3}\\[6pt]
e^{i\beta_{m''} z_3} - r_{m''2} e^{-i\beta_{m''} z_3}\end{bmatrix}
\tag{4.28}
\end{aligned}
$$

$$
\begin{aligned}

E^+(z_4) + E^-(z_4)\end{bmatrix}

&\quad\times \begin{bmatrix}e^{i\beta_2 z_4}\\[6pt]
e^{i\beta_2 z_4}\end{bmatrix}
\tag{4.29}
\end{aligned}
$$
