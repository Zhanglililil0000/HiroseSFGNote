### 4.2. 双层薄膜（1/m'/m/2）中的薄膜内电场

从入射侧开始依次设薄膜厚度为 $h_{m'}$ 和 $h_m$，由式 (3.3) 等可得以下结果。

$$
\begin{aligned}
E^-(0) &= \frac{(r_{1m'} + r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{1m'} r_{m'm} + e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}} E^+(0) \tag{4.7a}
\end{aligned}
$$

$$

\begin{aligned}
\begin{bmatrix}E^+(z_1)\\[6pt]
E^-(z_1)\end{bmatrix}
&= \frac{1}{t_{m'1}} \begin{bmatrix}e^{i\beta_{m'} z_1} & r_{m'1} e^{i\beta_{m'} z_1}\\[6pt]
r_{m'1} e^{-i\beta_{m'} z_1} & e^{-i\beta_{m'} z_1}\end{bmatrix} \begin{bmatrix}E^+(0)\\[6pt]
E^-(0)\end{bmatrix} \\[8pt]
&= \frac{t_{1m'} E^+(0)}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}(1 + r_{m'm} r_{m2} e^{2i\beta_m h_m}) e^{i\beta_{m'} z_1}\\[6pt]
(r_{m'm} + r_{m2} e^{2i\beta_m h_m}) e^{2i\beta_{m'} h_{m'}} e^{-i\beta_{m'} z_1}\end{bmatrix} \tag{4.8a}
\end{aligned}

$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_2)\\[6pt]
E^-(z_2)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} e^{i\beta_{m'} h_{m'}} E^+(0)}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}e^{i\beta_m z_2}\\[6pt]
r_{m2} e^{2i\beta_m h_m} e^{-i\beta_m z_2}\end{bmatrix} \tag{4.9a}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_3)\\[6pt]
E^-(z_3)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{m2} e^{i\beta_{m'} h_{m'}} e^{i\beta_m h_m} E^+(0)}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}e^{i\beta_2 z_3}\\[6pt]
0\end{bmatrix} \tag{4.10a}
\end{aligned}
$$

因此，

$$
\begin{aligned}
\begin{bmatrix}E^+(0) + E^-(0)\\[6pt]
E^+(0) - E^-(0)\end{bmatrix}
&= \frac{E^+(0)}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}\\
&\quad\times \begin{bmatrix}(1 + r_{1m'})[(1 + r_{m'm} e^{2i\beta_{m'} h_{m'}}) + r_{m2}(r_{m'm} + e^{2i\beta_{m'} h_{m'}}) e^{2i\beta_m h_m}]\\[6pt]
(1 - r_{1m'})[(1 - r_{m'm} e^{2i\beta_{m'} h_{m'}}) + r_{m2}(r_{m'm} - e^{2i\beta_{m'} h_{m'}}) e^{2i\beta_m h_m}]\end{bmatrix} \tag{4.7b}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_1) + E^-(z_1)\\[6pt]
E^+(z_1) - E^-(z_1)\end{bmatrix}
&= \frac{t_{1m'} E^+(0)}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}\\
&\quad\times \begin{bmatrix}(1 + r_{m'm} r_{m2} e^{2i\beta_m h_m}) e^{i\beta_{m'} z_1} + (r_{m'm} + r_{m2} e^{2i\beta_m h_m}) e^{2i\beta_{m'} h_{m'}} e^{-i\beta_{m'} z_1}\\[6pt]
(1 + r_{m'm} r_{m2} e^{2i\beta_m h_m}) e^{i\beta_{m'} z_1} - (r_{m'm} + r_{m2} e^{2i\beta_m h_m}) e^{2i\beta_{m'} h_{m'}} e^{-i\beta_{m'} z_1}\end{bmatrix} \tag{4.8b}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_2) + E^-(z_2)\\[6pt]
E^+(z_2) - E^-(z_2)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} e^{i\beta_{m'} h_{m'}} E^+(0)}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}e^{i\beta_m z_2} + r_{m2} e^{2i\beta_m h_m} e^{-i\beta_m z_2}\\[6pt]
e^{i\beta_m z_2} - r_{m2} e^{2i\beta_m h_m} e^{-i\beta_m z_2}\end{bmatrix} \tag{4.9b}
\end{aligned}
$$

$$
\begin{aligned}
\begin{bmatrix}E^+(z_3) + E^-(z_3)\\[6pt]
E^+(z_3) - E^-(z_3)\end{bmatrix}
&= \frac{t_{1m'} t_{m'm} t_{m2} e^{i\beta_{m'} h_{m'}} e^{i\beta_m h_m} E^+(0)}{(1 + r_{1m'} r_{m'm} e^{2i\beta_{m'} h_{m'}}) + (r_{m'm} + r_{1m'} e^{2i\beta_{m'} h_{m'}}) r_{m2} e^{2i\beta_m h_m}}
\begin{bmatrix}e^{i\beta_2 z_3}\\[6pt]
e^{i\beta_2 z_3}\end{bmatrix} \tag{4.10b}
\end{aligned}
$$
