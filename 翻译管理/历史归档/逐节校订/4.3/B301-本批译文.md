#### 反对称($b_1$)振动模式

在原文条件$\beta_{\xi\xi\zeta}\gg\beta_{\xi\xi\zeta}\sim\beta_{\eta\eta\zeta}$下：


$$
\beta_{caa} = \beta_{\zeta\zeta\zeta}(\cos\alpha - \cos^3\alpha) \approx \frac{2\sqrt{3}}{9}\beta_{\zeta\zeta\zeta}\ (T_d),\quad \frac{3}{8}\beta_{\zeta\zeta\zeta}\ (\text{sp}^2)
$$


**(ppp)**


$$
\begin{aligned}
\chi_{xxx} &= -\frac{1}{4}\beta_{caa}\{[(1 + \cos 2\phi)(\sin\theta - \sin^3\theta)(3\cos \chi + \cos 3\chi) + (1 - \cos 2\phi)\sin\theta(\cos \chi - \cos 3\chi)]
\\
&\quad - 2\sin 2\phi\sin\theta\cos\theta(\sin \chi + \sin 3\chi)\}
\end{aligned}
$$



$$
\chi_{xzz} = \frac{1}{2}\beta_{caa}[(1 + \cos 2\phi)(\sin\theta - 2\sin^3\theta)\cos \chi - \sin 2\phi\sin\theta\cos\theta\sin \chi]
$$



$$
\chi_{zxz} = \frac{1}{2}\beta_{caa}[(1 + \cos 2\phi)(\sin\theta - 2\sin^3\theta)\cos \chi - \sin 2\phi\sin\theta\cos\theta\sin \chi]
$$



$$
\chi_{zzx} = \beta_{caa}[(1 + \cos 2\phi)(\sin\theta - \sin^3\theta)\cos \chi - \sin 2\phi\sin\theta\cos\theta\sin \chi]
$$



$$
\begin{aligned}
\chi_{zxx} &= \frac{1}{4}\beta_{caa}\{2[\cos\theta(1 + \cos 2\phi\cos 2\chi) - (1 + \cos 2\phi)(\cos\theta - \cos^3\theta)(1 + \cos 2\chi)]
\\
&\quad + \sin 2\phi(1 - 3\cos^2\theta)\sin 2\chi\}
\end{aligned}
$$



$$
\begin{aligned}
\chi_{xzx} &= \frac{1}{4}\beta_{caa}\{2[\cos\theta(1 + \cos 2\phi\cos 2\chi) - (1 + \cos 2\phi)(\cos\theta - \cos^3\theta)(1 + \cos 2\chi)]
\\
&\quad + \sin 2\phi(1 - 3\cos^2\theta)\sin 2\chi\}
\end{aligned}
$$



$$
\chi_{xxz} = \frac{1}{2}\beta_{caa}[-(1 + \cos 2\phi)(\cos\theta - \cos^3\theta)(1 + \cos 2\chi) + \sin 2\phi\sin^2\theta\sin 2\chi]
$$



$$
\chi_{zzz} = \beta_{caa}(1 + \cos 2\phi)(\cos\theta - \cos^3\theta)
$$


**(spp)**


$$
\begin{aligned}
\chi_{yxx} &= \frac{1}{4}\beta_{caa}\{[(1 + \cos 2\phi)(\sin\theta - \sin^3\theta)(\sin \chi + \sin 3\chi) + (1 - \cos 2\phi)\sin\theta(\sin \chi - \sin 3\chi)]
\\
&\quad + 2\sin 2\phi\sin\theta\cos\theta\cos 3\chi\}
\end{aligned}
$$



$$
\chi_{yzz} = -\frac{1}{2}\beta_{caa}[(1 + \cos 2\phi)(\sin\theta - 2\sin^3\theta)\sin \chi + \sin 2\phi\sin\theta\cos\theta\cos \chi]
$$



$$
\chi_{yzx} = \frac{1}{4}\beta_{caa}\{2[(1 + \cos 2\phi)(\cos\theta - \cos^3\theta) - \cos 2\phi\cos\theta]\sin 2\chi + \sin 2\phi[-\sin^2\theta + (1 - 3\cos^2\theta)\cos 2\chi]\}
$$



$$
\chi_{yxz} = \frac{1}{2}\beta_{caa}[(1 + \cos 2\phi)(\cos\theta - \cos^3\theta)\sin 2\chi + \sin 2\phi\sin^2\theta\cos 2\chi]
$$


**(ssp)**


$$
\begin{aligned}
\chi_{yyx} &= \frac{1}{4}\beta_{caa}\{[-(1 + \cos 2\phi)(\sin\theta - \sin^3\theta) + (1 - \cos 2\phi)\sin\theta](\cos \chi - \cos 3\chi)
\\
&\quad + 2\sin 2\phi\sin\theta\cos\theta(\sin \chi - \sin 3\chi)\}
\end{aligned}
$$



$$
\chi_{yyz} = -\frac{1}{2}\beta_{caa}[(1 + \cos 2\phi)(\cos\theta - \cos^3\theta)(1 - \cos 2\chi) + \sin 2\phi\sin^2\theta\sin 2\chi]
$$


**(psp)**


$$
\begin{aligned}
\chi_{xyx} &= \frac{1}{4}\beta_{caa}\{[(1 + \cos 2\phi)(\sin\theta - \sin^3\theta)(\sin \chi + \sin 3\chi) + 1 - \cos 2\phi)\sin\theta(\sin \chi - \sin 3\chi)]
\\
&\quad + 2\sin 2\phi\sin\theta\cos\theta\cos 3\chi\}
\end{aligned}
$$



$$
\chi_{zyz} = -\frac{1}{2}\beta_{caa}[(1 + \cos 2\phi)(\sin\theta - 2\sin^3\theta)\sin \chi + \sin 2\phi\sin\theta\cos\theta\cos \chi]
$$



$$
\chi_{xyz} = \frac{1}{2}\beta_{caa}[(1 + \cos 2\phi)(\cos\theta - \cos^3\theta)\sin 2\chi + \sin 2\phi\sin^2\theta\cos 2\chi]
$$



$$
\chi_{zyx} = \frac{1}{4}\beta_{caa}\{2[(1 + \cos 2\phi)(\cos\theta - \cos^3\theta) - \cos 2\phi\cos\theta]\sin 2\chi + \sin 2\phi[-\sin^2\theta + (1 - 3\cos^2\theta)\cos 2\chi]\}
$$


**(sps)**


$$
\begin{aligned}
\chi_{yxy} &= -\frac{1}{4}\beta_{caa}\{[-(1 + \cos 2\phi)(\sin\theta - \sin^3\theta)(\cos \chi - \cos 3\chi) + (1 - \cos 2\phi)\sin\theta(\cos \chi + \cos 3\chi)]
\\
&\quad + 2\sin 2\phi\sin\theta\cos\theta\sin 3\chi\}
\end{aligned}
$$



$$
\begin{aligned}
\chi_{yzy} &= \frac{1}{4}\beta_{caa}\{2[\cos\theta(1 - \cos 2\phi\cos 2\chi) - (1 + \cos 2\phi)(\cos\theta - \cos^3\theta)(1 - \cos 2\chi)]
\\
&\quad - \sin 2\phi(1 - 3\cos^2\theta)\sin 2\chi\}
\end{aligned}
$$

$$
\begin{aligned}
\chi_{xxy} &= \frac{1}{4}\beta_{caa}\{[(1 + \cos 2\phi)(\sin\theta - \sin^3\theta) - (1 - \cos 2\phi)\sin\theta](\sin \chi + \sin 3\chi)
\\
&\quad + 2\sin 2\phi\sin\theta\cos\theta(\cos \chi + \cos 3\chi)\}

$$
\chi_{zzy} = -\beta_{caa}[(1 + \cos 2\phi)(\sin\theta - 2\sin^3\theta)\sin \chi + \sin 2\phi\sin\theta\cos\theta\cos \chi]
$$



$$
\chi_{zxy} = \frac{1}{4}\beta_{caa}\{2[(1 + \cos 2\phi)(\cos\theta - \cos^3\theta) - \cos 2\phi\cos\theta]\sin 2\chi + \sin 2\phi[\sin^2\theta + (1 - 3\cos^2\theta)\cos 2\chi]\}

$$
\chi_{xzy} = \frac{1}{4}\beta_{caa}\{2[(1 + \cos 2\phi)(\cos\theta - \cos^3\theta) - \cos 2\phi\cos\theta]\sin 2\chi + \sin 2\phi[\sin^2\theta + (1 - 3\cos^2\theta)\cos 2\chi]\}

$$
\begin{aligned}
\chi_{xyy} &= -\frac{1}{4}\beta_{caa}\{[-(1 + \cos 2\phi)(\sin\theta - \sin^3\theta)(\cos \chi - \cos 3\chi) + (1 - \cos 2\phi)\sin\theta(\cos \chi + \cos 3\chi)]

$$
\chi_{xxx} + \chi_{yyx} + \chi_{zzx} = \chi_{xxy} + \chi_{yyy} + \chi_{zzy} = \chi_{xxz} + \chi_{yyz} + \chi_{zzz} = 0
$$
