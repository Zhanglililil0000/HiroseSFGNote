$$
\begin{aligned}
E_{\mathrm{vis},\alpha}(z)
&=E^0_{\mathrm{vis},\alpha}
\frac{t_{1m,\mathrm{vis},\alpha}}
{1+r_{1m,\mathrm{vis},\alpha}r_{m2,\mathrm{vis},\alpha}e^{2i\beta_{m,\mathrm{vis}}h_m}}\\
&\quad\times\exp\!\left[-i\,2(n+1)k_{m,\mathrm{vis}}h_m\tan\theta_{m,\mathrm{SF}}\sin\theta_{m,\mathrm{vis}}\right]\\
&\quad\times\Bigl\{\exp\!\left[izk_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}
\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\right]\\
&\qquad+r_{m2,\mathrm{vis},\alpha}e^{2i\beta_{m,\mathrm{vis}}h_m}
\exp\!\left[-izk_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}
\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\right]\Bigr\}.
\end{aligned}\tag{4.19}
$$

$$
\begin{aligned}
E_{\mathrm{ir},\beta}(z)
&=E^0_{\mathrm{ir},\beta}
\frac{t_{1m,\mathrm{ir},\beta}}
{1+r_{1m,\mathrm{ir},\beta}r_{m2,\mathrm{ir},\beta}e^{2i\beta_{m,\mathrm{ir}}h_m}}\\
&\quad\times\exp\!\left[-i\,2(n+1)k_{m,\mathrm{ir}}h_m\tan\theta_{m,\mathrm{SF}}\sin\theta_{m,\mathrm{ir}}\right]\\
&\quad\times\Bigl\{\exp\!\left[izk_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}
\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\right]\\
&\qquad+r_{m2,\mathrm{ir},\beta}e^{2i\beta_{m,\mathrm{ir}}h_m}
\exp\!\left[-izk_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}
\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\right]\Bigr\}.
\end{aligned}\tag{4.20}
$$

$$
\begin{aligned}
&E_{\mathrm{vis},\alpha}(z)E_{\mathrm{ir},\beta}(z)\\
&=E^0_{\mathrm{vis},\alpha}E^0_{\mathrm{ir},\beta}
\frac{t_{1m,\mathrm{vis},\alpha}t_{1m,\mathrm{ir},\beta}}
{\left(1+r_{1m,\mathrm{vis},\alpha}r_{m2,\mathrm{vis},\alpha}e^{2i\beta_{m,\mathrm{vis}}h_m}\right)
\left(1+r_{1m,\mathrm{ir},\beta}r_{m2,\mathrm{ir},\beta}e^{2i\beta_{m,\mathrm{ir}}h_m}\right)}\\
&\quad\times\exp\!\left[-i(2n+1)h_m\tan\theta_{m,\mathrm{SF}}
\left(k_{m,\mathrm{vis}}\sin\theta_{m,\mathrm{vis}}+k_{m,\mathrm{ir}}\sin\theta_{m,\mathrm{ir}}\right)\right]\\
&\quad\times\Biggl\{\exp\!\Bigl[iz\Bigl(
k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad+k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\\
&\qquad+r_{m2,\mathrm{ir},\beta}e^{2i\beta_{m,\mathrm{ir}}h_m}
\exp\!\Bigl[iz\Bigl(
k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad-k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\\
&\qquad+r_{m2,\mathrm{vis},\alpha}e^{2i\beta_{m,\mathrm{vis}}h_m}
\exp\!\Bigl[iz\Bigl(
-k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad+k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\\
&\qquad+r_{m2,\mathrm{vis},\alpha}r_{m2,\mathrm{ir},\beta}
e^{2i(\beta_{m,\mathrm{vis}}+\beta_{m,\mathrm{ir}})h_m}\\
&\qquad\qquad\times\exp\!\Bigl[iz\Bigl(
-k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad-k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\Biggr\}.
\end{aligned}\tag{4.21}
$$
