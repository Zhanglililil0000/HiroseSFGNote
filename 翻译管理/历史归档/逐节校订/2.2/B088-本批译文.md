$$
\begin{aligned}
P_\alpha(h_m^+)
&=\sum_{\beta,\gamma}\chi_{\alpha\beta\gamma}
E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma}\\
&\quad\times\frac{
t_{1m,\mathrm{vis},\beta}t_{m2,\mathrm{vis},\beta}
t_{1m,\mathrm{ir},\gamma}t_{m2,\mathrm{ir},\gamma}
e^{i(\beta_{m,\mathrm{ir}}+\beta_{m,\mathrm{vis}})h_m}}
{\left(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{wvis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m}\right)
\left(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m}\right)}\\
&\quad\times\exp\!\left[i(2n+1)h_m\tan\theta_{m,\mathrm{SF}}
\left(-k_{m,\mathrm{ir}}\sin\theta_{m,\mathrm{ir}}-k_{m,\mathrm{vis}}\sin\theta_{m,\mathrm{vis}}\right)\right]\\
&=P_\alpha(h_m^+,n=0)
\exp\!\left[2inh_m\tan\theta_{m,\mathrm{SF}}
\left(-k_{m,\mathrm{ir}}\sin\theta_{m,\mathrm{ir}}-k_{m,\mathrm{vis}}\sin\theta_{m,\mathrm{vis}}\right)\right].
\end{aligned}\tag{4.28a}
$$
