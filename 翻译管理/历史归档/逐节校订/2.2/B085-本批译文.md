$$
\begin{aligned}
P_\alpha(0^-,n)
&=\sum_{\beta,\gamma}\chi_{\alpha\beta\gamma}
E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma}\\
&\quad\times\frac{
\left(1+r_{1m,\mathrm{vis},\beta}\right)
\left(1+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m}\right)
\left(1+r_{1m,\mathrm{ir},\gamma}\right)
\left(1+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m}\right)}
{\left(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m}\right)
\left(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m}\right)}\\

\left(-k_{m,\mathrm{ir}}\sin\theta_{m,\mathrm{ir}}-k_{m,\mathrm{vis}}\sin\theta_{m,\mathrm{vis}}\right)\right]\Bigr\}\\
&=P_\alpha(0^-,n=0)
\exp\!\left[i\,2nh_m\tan\theta_{m,\mathrm{SF}}

\end{aligned}\tag{4.26a}
