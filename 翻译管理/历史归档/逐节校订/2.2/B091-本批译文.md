$$
\begin{aligned}
P_\alpha(z)
&=\sum_{\beta,\gamma}\chi_{\alpha\beta\gamma}
E^0_{\mathrm{vis},\beta}E^0_{\mathrm{ir},\gamma}\\
&\quad\times\frac{t_{1m,\mathrm{vis},\beta}t_{1m,\mathrm{ir},\gamma}}
{\left(1+r_{1m,\mathrm{vis},\beta}r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{ir}}h_m}\right)
\left(1+r_{1m,\mathrm{ir},\gamma}r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{vis}}h_m}\right)}\\
&\quad\times\exp\!\left[-2ih_m\tan\theta_{m,\mathrm{SF}}
\left(k_{m,\mathrm{vis}}\sin\theta_{m,\mathrm{vis}}+k_{m,\mathrm{ir}}\sin\theta_{m,\mathrm{ir}}\right)\right]\\
&\quad\times\Biggl\{\exp\!\Bigl[iz\Bigl(
k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad+k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\\
&\qquad+r_{m2,\mathrm{ir},\gamma}e^{2i\beta_{m,\mathrm{ir}}h_m}
\exp\!\Bigl[iz\Bigl(
k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad-k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\\
&\qquad+r_{m2,\mathrm{vis},\beta}e^{2i\beta_{m,\mathrm{vis}}h_m}
\exp\!\Bigl[iz\Bigl(
-k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad+k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1+\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\\
&\qquad+r_{m2,\mathrm{vis},\beta}r_{m2,\mathrm{ir},\gamma}
e^{2i(\beta_{m,\mathrm{vis}}+\beta_{m,\mathrm{ir}})h_m}\\
&\qquad\qquad\times\exp\!\Bigl[iz\Bigl(
-k_{m,\mathrm{vis}}\cos\theta_{m,\mathrm{vis}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{vis}}\right)\\
&\qquad\qquad\qquad-k_{m,\mathrm{ir}}\cos\theta_{m,\mathrm{ir}}\left(1-\tan\theta_{m,\mathrm{SF}}\tan\theta_{m,\mathrm{ir}}\right)\Bigr)\Bigr]\Biggr\}\\
&\quad\times\exp\!\left[-i\,2nh_m\tan\theta_{m,\mathrm{SF}}
\left(k_{m,\mathrm{vis}}\sin\theta_{m,\mathrm{vis}}+k_{m,\mathrm{ir}}\sin\theta_{m,\mathrm{ir}}\right)\right]\\
&=P_\alpha(z,n=0)
\exp\!\left[-i\,2nh_m\tan\theta_{m,\mathrm{SF}}
\left(k_{m,\mathrm{vis}}\sin\theta_{m,\mathrm{vis}}+k_{m,\mathrm{ir}}\sin\theta_{m,\mathrm{ir}}\right)\right].
\end{aligned}\tag{4.30a}
$$
