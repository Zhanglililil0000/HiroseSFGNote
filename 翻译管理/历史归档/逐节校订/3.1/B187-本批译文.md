**m'/m 界面的 m' 侧极化产生的和频：**

（a）反射方向

在文件「一般式」的 (2.1) 式和 (2.3) 式中，令 $z_1 = h_{m'}$、$a_0^{*} = 1$、$a_0 = a$。

$$
\begin{aligned}
&E^{-}_{1}(0^{-})_{\mathrm{net}} \\ &= \frac{a\,t_{m'1}[(r_{m'm} + r_{m2}b^{2})L^{+}_{m'/m'} + (1+ r_{mm'}r_{m2}b^{2})L^{-}_{m'/m'}]}{1 + r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_1 = h_{m'})
\end{aligned}
$$

其中，
$$
\begin{aligned}
&t_{m'1,p}L^{+}_{m'/m',px} \\ &= t_{m'1,p}L^{-}_{m'/m',px} = L^{-}_{1/m',px},\quad t_{m'1,s}L^{+}_{m'/m',sy} = t_{m'1,s}L^{-}_{m'/m',sy} = L^{-}_{1/m',sy}
\end{aligned}
$$
$$
\begin{aligned}
&t_{m'1,p}L^{+}_{m'/m',pz} \\ &= t_{m'1,p}L^{-}_{m'/m',pz} = -L^{-}_{1/m',px} \;(n_{m''} = n_{m'})
\end{aligned}
$$

因此，
$$
\begin{aligned}
&E^{-}_{1}(0^{-})_{\mathrm{net}} \\ &= \frac{a[(1 + r_{mm'}r_{m2}b^{2}) \pm (r_{m'm} + r_{m2}b^{2})]L^{-}_{1/m'}}{1 + r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_1 = h_{m'})
\end{aligned}
$$
$$
\begin{aligned}
&= \frac{a(1 \pm r_{mm'})(1 \pm r_{m2}b^{2})L^{-}_{1/m'}}{1+ r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_1 = h_{m'})
\end{aligned}
$$

（x 和 y 分量取上符号，z 分量取下符号）

参照 (2.23) 式，

$$
\begin{aligned}
&E^{-}_{1,p}(0^{-})_{\mathrm{net}} \\ &= \sum_{\alpha,\beta} E^{0}_{vis,\alpha}E^{0}_{ir,\beta}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',vis,\alpha}e^{i\beta_{m',vis}h_{m'}}(1+ r_{m'm,vis,\alpha})(1+ r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}})}{{\begin{aligned}&1\\&+ r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&+ (r_{m'm,vis,\alpha} + r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}\end{aligned}}}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',ir,\beta}e^{i\beta_{m',ir}h_{m'}}(1+ r_{m'm,ir,\beta})(1+ r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}})}{1+ r_{1m',ir,\beta}r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}} + (r_{m'm,ir,\beta} + r_{1m',ir,\beta}e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}}}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times\Bigg\{
\frac{e^{i\beta_{m',SF}h_{m'}}[(1+ r_{mm',SF,p})(1+ r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}})]L^{-}_{1/m',px}\chi_{x\alpha\beta}}{1+ r_{1m',SF,p}r_{m'm,SF,p}e^{2i\beta_{m',SF}h_{m'}} + (r_{m'm,SF,p} + r_{1m',SF,p}e^{2i\beta_{m',SF}h_{m'}})r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}}
\end{aligned}
$$
$$
\begin{aligned}
&+
\frac{e^{i\beta_{m',SF}h_{m'}}[(1- r_{mm',SF,p})(1- r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}})]L^{-}_{1/m',pz}\chi_{z\alpha\beta}}{1+ r_{1m',SF,p}r_{m'm,SF,p}e^{2i\beta_{m',SF}h_{m'}} + (r_{m'm,SF,p} + r_{1m',SF,p}e^{2i\beta_{m',SF}h_{m'}})r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}}
\Bigg\}

\end{aligned}\tag{2.39a}
$$

$$
\begin{aligned}
&E^{-}_{1,s}(0^{-})_{\mathrm{net}} \\ &= \sum_{\alpha,\beta} E^{0}_{vis,\alpha}E^{0}_{ir,\beta}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',vis,\alpha}e^{i\beta_{m',vis}h_{m'}}(1+ r_{m'm,vis,\alpha})(1+ r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}})}{{\begin{aligned}&1\\&+ r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&+ (r_{m'm,vis,\alpha} + r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}\end{aligned}}}
\end{aligned}
$$

$$
\begin{aligned}

\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{e^{i\beta_{m',SF}h_{m'}}(1+ r_{mm',SF,s})(1+ r_{m2,SF,s}e^{2i\beta_{m,SF}h_{m}})L^{-}_{1/m',sy}\chi_{y\alpha\beta}}{1+ r_{1m',SF,s}r_{m'm,SF,s}e^{2i\beta_{m',SF}h_{m'}} + (r_{m'm,SF,s} + r_{1m',SF,s}e^{2i\beta_{m',SF}h_{m'}})r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}}

\end{aligned}\tag{2.39b}
$$


（b）透射方向

在文件「一般式」的 (2.2) 式和 (2.4) 式中，令 $z_1 = h_{m'}$、$a_0^{*} = 1$、$a_0 = a$。

$$
\begin{aligned}
&E^{+}_{2}(h_{m'}+h_{m}^{+})_{\mathrm{net}} \\ &= \frac{t_{m2}t_{m'm}b(L^{+}_{m'/m'} + r_{m'1}a^{2}L^{-}_{m'/m'})}{1+ r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_1 = h_{m'})
\end{aligned}
$$

其中，
$$
\begin{aligned}
&t_{m'm,p}L^{+}_{m'/m',px} \\ &= L^{+}_{m'/m,px},\; t_{m'm,s}L^{+}_{m'/m',sy} = L^{+}_{m'/m,sy},\; t_{m'm,p}L^{+}_{m'/m',pz} = -t_{m'm,p}L^{-}_{m'/m',pz} = L^{+}_{m'/m,pz} \;(n_{m''} = n_{m'})
\end{aligned}
$$

因此，
$$
\begin{aligned}
&E^{+}_{2}(h_{m'}+h_{m}^{+})_{\mathrm{net}} \\ &= \frac{t_{m2}b(1 \pm r_{m'1}a^{2})L^{+}_{m'/m}}{1+ r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_1 = h_{m'})
\end{aligned}
$$
