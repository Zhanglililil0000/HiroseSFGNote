**1/m′界面m′侧的极化产生的和频光：**

（a）反射方向。

在文件《一般式》的(2.1)和(2.3)式中，令 $z_1=0$、$a_0^*=a$、$a_0=1$。

$$
\begin{aligned}
&E_1^-(0^-)_{net}\\ &=\frac{t_{m\prime1}[(r_{m\prime m}+r_{m2}b^2)a^2L^+_{m\prime/m\prime}+(1+r_{m\prime m}r_{m2}b^2)L^-_{m\prime/m\prime}]}{1+r_{1m\prime}r_{m\prime m}a^2+(r_{m\prime m}+r_{1m\prime}a^2)r_{m2}b^2}P^*(z_1=0)
\end{aligned}
$$

其中，

$$
\begin{aligned}&t_{1m\prime,p}L^+_{m\prime/m\prime,px}\\ &=t_{1m\prime,p}L^-_{m\prime/m\prime,px}=L^-_{1/m\prime,px},\\t_{1m\prime,s}L^+_{m\prime/m\prime,sy}&=t_{1m\prime,s}L^-_{m\prime/m\prime,sy}=L^-_{1/m\prime,sy},\\t_{1m\prime,p}L^+_{m\prime/m\prime,pz}&=-t_{1m\prime,p}L^-_{m\prime/m\prime,pz}=L^-_{1/m\prime,pz}\quad(n_{m\prime\prime}=n_{m\prime}).\end{aligned}
$$

因此，

$$
\begin{aligned}
&E_1^-(0^-)_{net}\\ &=\frac{t_{m\prime1}[(1+r_{m\prime m}r_{m2}b^2)\pm(r_{m\prime m}+r_{m2}b^2)a^2]}{1+r_{1m\prime}r_{m\prime m}a^2+(r_{m\prime m}+r_{1m\prime}a^2)r_{m2}b^2}L^-_{1/m}P^*(z_1=0)
\end{aligned}
$$

上符号用于x、y分量，下符号用于z分量。由式(2.19)，

$$
\begin{aligned}
&E^{-}_{1}(0^{-})_{\mathrm{net},p} \\ &= \sum_{\alpha,\beta} E^{0}_{vis,\alpha}E^{0}_{ir,\beta}




\\[4pt]&\times

\frac{t_{1m',vis,\alpha}[(1 + r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}) + (r_{m'm,vis,\alpha} + e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}]}{{\begin{aligned}&1\\&+ r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&+ (r_{m'm,vis,\alpha} + r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}\end{aligned}}}




\\[4pt]&\times

\frac{t_{1m',ir,\beta}[(1+ r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}}) + (r_{m'm,ir,\beta} + e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}}]}{1+ r_{1m',ir,\beta}r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}} + (r_{m'm,ir,\beta} + r_{1m',ir,\beta}e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}}}




\\[4pt]&\times[

\frac{(1 + r_{m'm,SF,p}r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}) + (r_{m'm,SF,p} + r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}})e^{2i\beta_{m',SF}h_{m'}}}{1+ r_{1m',SF,p}r_{m'm,SF,p}e^{2i\beta_{m',SF}h_{m'}} + (r_{m'm,SF,p} + r_{1m',SF,p}e^{2i\beta_{m',SF}h_{m'}})r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}} L^{-}_{1/m,px}\chi_{x\alpha\beta}






\\&\quad+ \frac{(1 + r_{m'm,SF,p}r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}) - (r_{m'm,SF,p} + r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}})e^{2i\beta_{m',SF}h_{m'}}}{1 + r_{1m',SF,p}r_{m'm,SF,p}e^{2i\beta_{m',SF}h_{m'}} + (r_{m'm,SF,p} + r_{1m',SF,p}e^{2i\beta_{m',SF}h_{m'}})r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}} L^{-}_{1/m,pz}\chi_{z\alpha\beta}] \quad (n_{m''} = n_{m'}) 
\end{aligned}\tag{2.33a}
$$

$$
\begin{aligned}
&E^{-}_{1}(0^{-})_{\mathrm{net},s} \\ &= \sum_{\alpha,\beta} E^{0}_{vis,\alpha}E^{0}_{ir,\beta}




\\[4pt]&\times

\frac{t_{1m',vis,\alpha}[(1 + r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}) + (r_{m'm,vis,\alpha} + e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}]}{{\begin{aligned}&1\\&+ r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&+ (r_{m'm,vis,\alpha} + r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}\end{aligned}}}




\\[4pt]&\times

\frac{t_{1m',ir,\beta}[(1+ r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}}) + (r_{m'm,ir,\beta} + e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}}]}{1+ r_{1m',ir,\beta}r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}} + (r_{m'm,ir,\beta} + r_{1m',ir,\beta}e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}}}

\end{aligned}

$$
