**m/2 界面的 2 侧极化产生的和频：**

（a）反射方向

仅有 $E^{-}(h_{m}^{+})$ 进入薄膜内部并穿透到另一侧。在文件「一般式」的 (2.7) 式中，令 $z_2 = h_m$、$b_0^{*} = 1$、$b_0 = b$。

$$
\begin{aligned}
&E^{-}_{1}(0^{-})_{\mathrm{net}} \\ &= \frac{t_{2m}t_{m'1}b^{2}(r_{m'm} + r_{1m'}a^{2})L^{-}_{2/2}}{1 + r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_2 = h_{m}^{+})
\end{aligned}
$$

其中，
$$
\begin{aligned}
&t_{2m,p}L^{-}_{2/2,px} \\ &= L^{-}_{m/2,px},\quad t_{2m,s}L^{-}_{2/2,sy} = L^{-}_{m/2,sy},\quad t_{2m,p}L^{-}_{2/2,pz} = L^{-}_{m/2,pz} \;(n_{m''} = n_2)
\end{aligned}
$$

因此，
$$
\begin{aligned}
&E^{-}_{1}(0^{-})_{\mathrm{net}} \\ &= \frac{t_{m'1}b^{2}(r_{m'm} + r_{1m'}a^{2})L^{-}_{m/2}}{1 + r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_2 = h_{m}^{+})
\end{aligned}
$$

参照 (2.20) 式，

$$
\begin{aligned}
&E^{-}_{1,p}(0^{-})_{\mathrm{net}} \\ &= \sum_{\alpha,\beta} E^{0}_{vis,\alpha}E^{0}_{ir,\beta}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',vis,\alpha}t_{m'm,vis,\alpha}t_{m2,vis,\alpha}e^{i(\beta_{m',vis}h_{m'}+\beta_{m,vis}h_{m})}}{{\begin{aligned}&1\\&+ r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&+ (r_{m'm,vis,\alpha} + r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}\end{aligned}}}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',ir,\beta}t_{m'm,ir,\beta}t_{m2,ir,\beta}e^{i(\beta_{m',ir}h_{m'}+\beta_{m,ir}h_{m})}}{1+ r_{1m',ir,\beta}r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}} + (r_{m'm,ir,\beta} + r_{1m',ir,\beta}e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}}}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{2m,SF,p}t_{m'1,SF,p}e^{2i\beta_{m,SF}h_{m}}(r_{m'm,SF,p} + r_{1m',SF,p}e^{2i\beta_{m',SF}h_{m'}})[L^{-}_{m/2,px}\chi_{x\alpha\beta} + L^{-}_{m/2,pz}\chi_{z\alpha\beta}]}{1 + r_{1m',SF,p}r_{m'm,SF,p}e^{2i\beta_{m',SF}h_{m'}} + (r_{m'm,SF,p} + r_{1m',SF,p}e^{2i\beta_{m',SF}h_{m'}})r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}}

\end{aligned}\tag{2.35a}
$$

$$
\begin{aligned}
&E^{-}_{1,s}(0^{-})_{\mathrm{net}} \\ &= \sum_{\alpha,\beta} E^{0}_{vis,\alpha}E^{0}_{ir,\beta}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',vis,\alpha}t_{m'm,vis,\alpha}t_{m2,vis,\alpha}e^{i(\beta_{m',vis}h_{m'}+\beta_{m,vis}h_{m})}}{{\begin{aligned}&1\\&+ r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&+ (r_{m'm,vis,\alpha} + r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}\end{aligned}}}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',ir,\beta}t_{m'm,ir,\beta}t_{m2,ir,\beta}e^{i(\beta_{m',ir}h_{m'}+\beta_{m,ir}h_{m})}}{1+ r_{1m',ir,\beta}r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}} + (r_{m'm,ir,\beta} + r_{1m',ir,\beta}e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_{m}}}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{2m,SF,s}t_{m'1,SF,s}(r_{m'm,SF,s} + r_{1m',SF,s}e^{2i\beta_{m',SF}h_{m'}})e^{2i\beta_{m,SF}h_{m}} L^{-}_{m/2,sy}\chi_{y\alpha\beta}}{1+ r_{1m',SF,p}r_{m'm,SF,p}e^{2i\beta_{m',SF}h_{m'}} + (r_{m'm,SF,p} + r_{1m',SF,p}e^{2i\beta_{m',SF}h_{m'}})r_{m2,SF,p}e^{2i\beta_{m,SF}h_{m}}}

\end{aligned}\tag{2.35b}
$$

（b）透射方向

除了零次光产生的 $E^{+}(z_2 = h_m^{+}) + r_{2m}E^{-}(z_2 = h_m^{+})$ 之外，还需加上进入内部的光经多次反射后出射的贡献，即在文件「一般式」的 (2.8) 式中，将 $L^{-}_{m/m}P(z_2)$ 替换为 $t_{1m'}L^{-}_{2/2}P(z_2 = h_m^{+})$ 后所得的部分。此外，令 $z_2 = h_m$、$b_0^{*} = 1$、$b_0 = b$。
$$
\begin{aligned}
&E^{+}_{2}(h_{m'}+h_{m}^{+})_{\mathrm{net}} \\ &= \{L^{+}_{2/2} + L^{-}_{2/2}[r_{2m} - \frac{t_{2m}t_{m2}b^{2}(r_{m'm} + r_{1m'}a^{2})}{1+ r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}}]\}P^{*}(z_2 = h_{m}^{+})
\end{aligned}
$$

$$
\begin{aligned}
&= \frac{(1+ r_{1m'}r_{m'm}a^{2})(L^{+}_{2/2} + r_{2m}L^{-}_{2/2}) + (r_{m'm} + r_{1m'}a^{2})b^{2}(r_{m2}L^{+}_{2/2} + L^{-}_{2/2})}{1 + r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_2 = h_{m}^{+})
\end{aligned}
$$

其中，

$$
\begin{aligned}
&(L^{+}_{2/2,px} + r_{2m,p}L^{-}_{2/2,px}) \\ &= (r_{m2,p}L^{+}_{2/2,px} + L^{-}_{2/2,px}) = L^{+}_{2/m,px}
\end{aligned}
$$
$$
\begin{aligned}
&(L^{+}_{2/2,sy} + r_{2m,s}L^{-}_{2/2,sy}) \\ &= (r_{m2,s}L^{+}_{2/2,sy} + L^{-}_{2/2,sy}) = L^{+}_{2/m,sy}
\end{aligned}
$$
$$
\begin{aligned}
&(L^{+}_{2/2,pz} + r_{2m,p}L^{-}_{2/2,pz}) \\ &= -(r_{m2,p}L^{+}_{2/2,pz} + L^{-}_{2/2,pz}) = L^{+}_{2/m,pz} \;(n_{m''} = n_2)
\end{aligned}
$$

因此，

$$
\begin{aligned}
&E^{+}_{2}(h_{m'}+h_{m}^{+})_{\mathrm{net}} \\ &= \frac{(1+ r_{1m'}r_{m'm}a^{2}) \pm (r_{m'm} + r_{1m'}a^{2})b^{2}}{1+ r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} L^{+}_{2/m}P^{*}(z_2 = h_{m}^{+})
\end{aligned}
$$

（x 和 y 分量取上符号，z 分量取下符号）

参照 (2.20) 式，

$$
\begin{aligned}
&E^{+}_{2,p}(h_{m'}+h_{m}^{+})_{\mathrm{net}} \\ &= \sum_{\alpha,\beta} E^{0}_{vis,\alpha}E^{0}_{ir,\beta}
\end{aligned}
$$

$$
\begin{aligned}
&\\[4pt]&\times \frac{t_{1m',vis,\alpha}t_{m'm,vis,\alpha}t_{m2,vis,\alpha}e^{i(\beta_{m',vis}h_{m'}+\beta_{m,vis}h_{m})}}{{\begin{aligned}&1\\&+ r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&+ (r_{m'm,vis,\alpha} + r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_{m}}\end{aligned}}}
\end{aligned}
$$
