**m/2界面的2侧：**

（a）：$E^+$ 源。

$$
\begin{aligned}
E_{vis,\alpha}(h_{m'}+h_m^+)E_{ir,\beta}(h_{m'}+h_m^+)
 &=E^0_{vis,\alpha}E^0_{ir,\beta}\\
 &\quad\times \frac{t_{1m',vis,\alpha}t_{m'm,vis,\alpha}t_{m2,vis,\alpha}e^{i(\beta_{m',vis}h_{m'}+\beta_{m,vis}h_m)}}{\begin{aligned}&1+r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&\quad+(r_{m'm,vis,\alpha}+r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_m}\end{aligned}}\\[6pt]
 &\quad\times \frac{t_{1m',ir,\beta}t_{m'm,ir,\beta}t_{m2,ir,\beta}e^{i(\beta_{m',ir}h_{m'}+\beta_{m,ir}h_m)}}{\begin{aligned}&1+r_{1m',ir,\beta}r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}}\\&\quad+(r_{m'm,ir,\beta}+r_{1m',ir,\beta}e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_m}\end{aligned}}\\[6pt]
 &\quad\times\exp\!\Bigl[2ni
 (h_m\tan\theta_{m,SF}+h_{m'}\tan\theta_{m',SF})\\
 &\hspace{7em}\times(-k_{m,ir}\sin\theta_{m,ir}
 -k_{m,vis}\sin\theta_{m,vis})\Bigr]
\end{aligned}
\tag{2.3}
$$

（b）：$E^-$ 源（$n>0$ 时，经反射）。

$$
\begin{aligned}
E_{vis,\alpha}(h_{m'}+h_m^+)E_{ir,\beta}(h_{m'}+h_m^+)
 &=E^0_{vis,\alpha}E^0_{ir,\beta}\\
 &\quad\times \frac{t_{1m',vis,\alpha}t_{m'm,vis,\alpha}t_{m2,vis,\alpha}e^{i(\beta_{m',vis}h_{m'}+\beta_{m,vis}h_m)}}{\begin{aligned}&1+r_{1m',vis,\alpha}r_{m'm,vis,\alpha}e^{2i\beta_{m',vis}h_{m'}}\\&\quad+(r_{m'm,vis,\alpha}+r_{1m',vis,\alpha}e^{2i\beta_{m',vis}h_{m'}})r_{m2,vis,\alpha}e^{2i\beta_{m,vis}h_m}\end{aligned}}\\[6pt]
 &\quad\times \frac{t_{1m',ir,\beta}t_{m'm,ir,\beta}t_{m2,ir,\beta}e^{i(\beta_{m',ir}h_{m'}+\beta_{m,ir}h_m)}}{\begin{aligned}&1+r_{1m',ir,\beta}r_{m'm,ir,\beta}e^{2i\beta_{m',ir}h_{m'}}\\&\quad+(r_{m'm,ir,\beta}+r_{1m',ir,\beta}e^{2i\beta_{m',ir}h_{m'}})r_{m2,ir,\beta}e^{2i\beta_{m,ir}h_m}\end{aligned}}\\[6pt]
 &\quad\times\exp\!\Bigl[i(2n+1)
 (h_m\tan\theta_{m,SF}+h_{m'}\tan\theta_{m',SF})\\
 &\hspace{7em}\times(-k_{m,ir}\sin\theta_{m,ir}
 -k_{m,vis}\sin\theta_{m,vis})\Bigr]
\end{aligned}
\tag{2.4}
$$
