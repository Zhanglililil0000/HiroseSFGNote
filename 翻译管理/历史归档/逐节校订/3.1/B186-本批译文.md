\begin{aligned}

$$

\end{aligned}

$$
\begin{aligned}

\begin{aligned}
&E^{+}_{2}(h_{m'}+h_{m}^{+})_{\mathrm{net}} \\ &= \frac{(1 + r_{1m'}r_{m'm}a^{2})L^{+}_{m/m} - (r_{m'm} + r_{1m'}a^{2})b^{2}L^{-}_{m/m}}{1 + r_{1m'}r_{m'm}a^{2} + (r_{m'm} + r_{1m'}a^{2})r_{m2}b^{2}} P^{*}(z_2 = h_{m}^{-})
\end{aligned}

其中，
$$
\begin{aligned}
